const MESURE = () => {
  const vis = e => { const r = e.getBoundingClientRect(); const s = getComputedStyle(e); return r.width > 0 && r.height > 0 && s.visibility !== 'hidden' && s.display !== 'none'; };
  // taille de texte pondérée par le nombre de caractères propres à chaque élément
  const classes = {}; let total = 0;
  document.querySelectorAll('#app *, .barre-navigation-fixe *').forEach(e => {
    if (!vis(e)) return;
    let n = 0; e.childNodes.forEach(c => { if (c.nodeType === 3) n += c.textContent.trim().length; });
    if (!n) return;
    const px = Math.round(parseFloat(getComputedStyle(e).fontSize) * 10) / 10;
    classes[px] = (classes[px] || 0) + n; total += n;
  });
  const petits = Object.entries(classes).filter(([k]) => +k < 14).reduce((a, [, v]) => a + v, 0);
  const btns = [...document.querySelectorAll('#app button, #app a.btn, .barre-navigation-fixe button, #app [role=button]')].filter(vis).map(e => {
    const r = e.getBoundingClientRect(); return { t: (e.innerText || e.getAttribute('aria-label') || '').trim().slice(0, 28), h: Math.round(r.height), w: Math.round(r.width), fs: Math.round(parseFloat(getComputedStyle(e).fontSize) * 10) / 10 };
  });
  const hs = btns.map(b => b.h).sort((a, b) => a - b);
  return {
    texteTotal: total, partTexteSous14px: total ? Math.round(1000 * petits / total) / 10 : 0,
    tailles: Object.entries(classes).sort((a, b) => +a[0] - +b[0]).map(([k, v]) => k + ':' + v).join(' '),
    nbBoutons: btns.length, boutonHauteurMin: hs[0] || null, boutonHauteurMediane: hs[Math.floor(hs.length / 2)] || null,
    boutonsSous40: btns.filter(b => b.h < 40).length, boutonsSous44: btns.filter(b => b.h < 44).length,
    petitsBoutons: btns.filter(b => b.h < 36).slice(0, 8), scrollHorizontal: document.documentElement.scrollWidth > innerWidth + 1,
    hauteurPage: document.documentElement.scrollHeight, corpsPx: getComputedStyle(document.body).fontSize
  };
};
